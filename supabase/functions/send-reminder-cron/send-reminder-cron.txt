import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const supabase = createClient(
  Deno.env.get('SUPABASE_URL')!,
  Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
)

function formatCurrency(value: number | null) {
  if (!value) return 'N/A'

  return value.toLocaleString('pt-BR', {
    style: 'currency',
    currency: 'BRL'
  })
}

function formatarEmailLembrete(cotacoes: any[]) {
  if (cotacoes.length === 0) {
    return null
  }

  let html = `
    <h1>Lembretes de Cotações</h1>
    <p>Eventos com 60 dias para acontecer:</p>
  `

  cotacoes.forEach((c) => {
    html += `
      <div style="
        border:1px solid #ccc;
        padding:15px;
        margin-bottom:10px;
      ">
        <h3>${c.codigo_cotacao}</h3>

        <p><strong>Cliente:</strong> ${c.nome || 'N/A'}</p>

        <p><strong>Telefone:</strong> ${c.telefone || 'N/A'}</p>

        <p><strong>Cidade:</strong> ${c.cidade}</p>

        <p>
          <strong>Preço PB:</strong>
          ${formatCurrency(c.price_pb)}
        </p>

        <p>
          <strong>Preço Colorido:</strong>
          ${formatCurrency(c.price_colorido)}
        </p>

        <a href="${c.whatsapp_link}">
          WhatsApp
        </a>
      </div>
    `
  })

  return html
}
